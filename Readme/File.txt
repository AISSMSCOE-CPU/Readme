#include <iostream>
#include <string>
using namespace std;

const int N = 5;

string places[N] = {
    "City", "Busstop", "Hospital", "Mall", "Garden"
};

int adj[N][N];
int count[N] = {0};
int visited[N] = {0};
void addRoad(int u, int v){
    adj[u][count[u]++] = v;
    adj[v][count[v]++]= u;
}

void DFS(int v){
    cout << places[v] << " ";
    visited[v] = 1;

    for(int i =0; i<count[v];i++){
        int next = adj[v][i];
        if(visited[next] == 0){
            DFS(next);
        }
    }

}

void BFS(int start){
    int q[N];
    int front = 0, rear = 0;
    visited[start] = 1;
    q[rear++] = start;

    while(front<rear){
        int v = q[front++];
        cout <<places[v] <<" ";
        for(int i=0; i<count[v] ; i++){
            int next = adj[v][i];
            if(visited[next] == 0){
            visited[next] = 1;
            q[rear++] = next;
}

        }
    }   

}

int main(){
    addRoad(0, 1);
    addRoad(0, 2);
    addRoad(1, 3);
    addRoad(2, 4);
    addRoad(3, 4);

    cout << "DFS: " << " ";
    DFS(0);

    for(int i =0;i<N; i++){
        visited[i] = 0;
    }
    cout << "BFS: " << " ";
    BFS(0);

    return 0;

}

#include <iostream>
#include <cctype>
using namespace std;

char stack[100];
int top = -1;

void push(char x){
    stack[++top] = x;

}

char pop(){
    return stack[top--];
}

int priority(char x){
    if(x == '*' || x == '/'){
        return 2;
    }
    else if(x == '+' || x == '-'){
        return 1;
    }
    else{
        return 0;
    }
}

int main(){
char infix[100],postfix[100];
int j=0;
cout << "Enter the infix expression: ";
cin >> infix;

for(int i =0; infix[i] != '\0';i++){
    char x = infix[i];
    if(isalnum(x)){
        postfix[j++] = x;
    }
    else if(x == '('){
        push(x);
    }
    else if( x ==')'){
        while(top != -1 && stack[top] != '('){
            postfix[j++] = pop();
        }
        if(top!= -1){
            pop();
        }
        
    }
    else{
        while(top !=-1 && priority(stack[top]) >= priority(x)){
            postfix[j++] = pop();
        }
        push(x);
    }


}
while(top != -1){
    postfix[j++] = pop();
}
postfix[j] = '\0';

cout<<"Postfix expression" << postfix << endl;

int value[100];
int t = -1;

cout<<"Enter the digits values for operands if any: ";

for(int i = 0; postfix[i] != '\0'; i++){
    char x = postfix[i];

    if(isdigit(x)){
        value[++t] = x - '0';
    }
    else if(isalpha(x)){
        int v;
        cout << x << " ";
        cin >> v;
        value[++t] = v;
    }
    else{
        int b = value[t--];
        int a = value[t--];

        switch(x){
            case '+': value[++t] = a+b; break;
            case '-': value[++t] = a-b; break;
            case '*': value[++t] = a*b; break;
            case '/':
            if(b==0){
                cout << "Invalid Input for B";
                return 0;
            }
            value[++t] = a/b; break;
            
        }
    }
}
cout << "Result: " << value[t] << endl;
return 0;
}

        rear = -1;
    }

    void placeorder(){
        int order;
        if((rear+1)%M == front){
            cout << "Queue is full...";
            return;
        }

        cout << "Enter the order number";
        cin >> order;

        if(front == -1){
            front =0;
        }

        rear = (rear+1)%M;
        queue[rear] = order;
        cout << "Order is placed";
    }

    void serveorder(){
        if(front==-1){
            cout << "No orders being served...";
            return;
        }
        cout << "Order Ready!!" << queue[front];
        if(front == rear){
            front = -1;
            rear = -1;
        }
        else{
            front = (front+1)%M;
        }
    }

    void display(){
        if(front == -1){
            cout << "Queue is empty...";
            return;
        }
        int i = front;
        for(int i =front ; ; i = (i+1)%M){
            cout << queue[i] << " ";
            if(i == rear){
                break;
            }
        }
        cout << endl;
    }

#include <iostream>
using namespace std;

struct Node{
    Node *left,*right;
    int data;
};

Node* createnode(int x){
    Node *temp = new Node;
    temp->data = x;
    temp->left = nullptr;
    temp->right = nullptr;
    return temp;
}

Node *insert(Node *root,int x){
    if(root == nullptr){
        return createnode(x);
    }
    if(x < root->data){
        root->left = insert(root->left , x);
    }
    else if(x>root->data){
        root->right = insert(root->right,x);
    }

    return root;
}

int height(Node *root){
    if(root == nullptr){
        return 0;
    }
    int l = height(root->left);
    int r = height(root->right);

    if(l>r){
        return l+1;
    }
    else{
        return r+1;
    }
}

int minimum(Node *root){
    if(root == nullptr){
        return 0;
    }

    while(root->left != nullptr){
        root = root->left;
    }
    return root->data;
}

void swaptree(Node *root){
    if(root == nullptr){
        return;
    }
    Node *temp = root->left;
    root->left = root->right;
    root->right = temp;

    swaptree(root->left);
    swaptree(root->right);
}

int search(Node *root, int x){
    if(root == nullptr){
        return 0;
    }
    if(x == root->data){
        return 1;
    }
    else if( x<root->data){
        return search(root->left,x);
    }
    else{
        return search(root->right,x);
    }
}

void inorder(Node *root){
    if(root != nullptr){
        inorder(root->left);
        cout << root->data << " ";
        inorder(root->right); 
    }
    
}
